## Hard logic
```ts
if (totalPages <= 7) return 1..totalPages;
const keep = sorted unique of [1, page - 1, page, page + 1, totalPages] within 1..totalPages;
let prev = 0;
for (const p of keep) {
  if (p - prev === 2) out.push(prev + 1);   // a one-page gap shows the number
  else if (p - prev > 2) out.push('gap');
  out.push(p); prev = p;
}
```
