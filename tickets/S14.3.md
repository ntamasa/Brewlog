## Hard logic
```ts
private readonly injector = inject(Injector);
private readonly newMatInput = viewChild('newInput', { read: MatInput });

constructor() {
  // #new-m has no ngControl, so MatInput never re-runs addErrorMatcher by itself:
  // re-run it whenever addError changes or the input (re)appears.
  effect(() => {
    this.addError();
    this.newMatInput()?.updateErrorState();
  });
}

// pending() is still 'add' inside next/error (the store's finalize runs right after them),
// so #new-m is still disabled there: focus it after the render that enables it again.
private refocusAddInput(): void {
  afterNextRender(() => this.newInput().nativeElement.focus(), { injector: this.injector });
}
```
