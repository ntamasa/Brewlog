## Hard logic
```ts
// Hold the POST 2.5 s, then let it reach the real backend.
await page.route("**/api/brews", async (route) => {
  if (route.request().method() !== "POST") return route.continue();
  await new Promise((resolve) => setTimeout(resolve, 2500));
  await route.continue();
});

// The browser's own serialisation of on-surface at 38%, for test 3.
const grey = await page.evaluate(() => {
  const probe = document.createElement("span");
  probe.style.color = "color-mix(in srgb, rgb(43, 29, 22) 38%, transparent)";
  document.body.append(probe);
  const color = getComputedStyle(probe).color;
  probe.remove();
  return color;
});
```
